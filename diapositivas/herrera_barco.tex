\begin{frame}{Transparencias durante el trabajo paralelo}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{block}{Transparencia}
        Varios nodos se coordinan por mensajes, pero el usuario ve una sola unidad.~\cite{hbFreitas2025}
      \end{block}

      \vspace{0.8em}
      {\small
      Procesos paralelos en la nube\\
      $\downarrow$ se usan como si fueran \textbf{locales}~\cite{hbArjona2023}\par}
    \end{column}

    \begin{column}{0.52\textwidth}
      \begin{block}{Qué se oculta}
        \small
        \begin{itemize}
          \item \textbf{Acceso:} misma operación local o remota.~\cite{hbArjona2023}
          \item \textbf{Ubicación:} dónde está el recurso.
          \item \textbf{Concurrencia:} actualizaciones simultáneas.~\cite{hbFreitas2025}
          \item \textbf{Replicación:} copias en varios sitios.~\cite{hbFreitas2025}
          \item \textbf{Fallos:} las réplicas mantienen el servicio.~\cite{hbFreitas2025}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Comunicación que coordina tareas}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Modelos}
        \small
        \begin{itemize}
          \item \textbf{Cliente-servidor:} solicitud y respuesta.~\cite{hbSimoes2026}
          \item \textbf{Peer-to-peer:} los nodos interactúan directamente.~\cite{hbSaleh2025}
          \item \textbf{Híbrido:} un coordinador central y nodos iguales.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Mecanismos}
        \small
        \begin{itemize}
          \item \textbf{Sockets:} canal TCP o UDP entre procesos.
          \item \textbf{RPC:} invocar un procedimiento remoto (gRPC).~\cite{hbSimoes2026}
          \item \textbf{Mensajería:} colas que guardan mensajes y desacoplan emisor y receptor.~\cite{hbSaleh2025}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
